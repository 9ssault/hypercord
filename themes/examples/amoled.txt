// language: JavaScript, file: themes/examples/amoled.txt
//
// The theme the client ships as its default: full AMOLED black. A theme is data,
// not code - the ten colour tokens themes/index.js accepts, each a six-digit hex
// value, and no lifecycle method at all (a theme that declares one is refused
// before it is installed, because themes have no hooks and no remote assets).
//
// The values are the same ones the desktop client's built-in theme uses (AMOLED_CSS
// in src/themes.js), so the two clients look like one product. Every surface is
// true black: the raised and recessed surfaces used to sit a hair above it so two
// surfaces had an edge between them, and the result was a theme that looked dark
// grey rather than black - the edges come from Discord's own 1px borders and the
// white-alpha hover fills instead. Text is left light for the same reason: the
// background is the theme.
export default {
  id: "example.amoled",
  kind: "theme",
  name: "AMOLED Black",
  version: "1.0.0",
  description: "True black on every surface, with edges drawn by Discord's own borders rather than by a lifted grey.",
  colors: {
    BACKGROUND_PRIMARY: "#000000",
    BACKGROUND_SECONDARY: "#000000",
    BACKGROUND_TERTIARY: "#000000",
    BACKGROUND_ACCENT: "#000000",
    TEXT_NORMAL: "#f2f3f5",
    TEXT_MUTED: "#a3a6aa",
    TEXT_LINK: "#7aa2ff",
    HEADER_PRIMARY: "#ffffff",
    BRAND_500: "#5865f2",
    STATUS_DANGER: "#f2555a"
  }
}
